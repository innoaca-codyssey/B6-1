-- Q10: 카테고리별 도서 대체 구입비를 합산합니다.
SELECT c.name, SUM(b.replacement_cost) AS total_cost FROM category c
INNER JOIN book b ON b.category_id = c.id
GROUP BY c.id, c.name ORDER BY total_cost DESC, c.id;
