-- Q09: 회원별 대여 건수를 집계합니다.
SELECT m.name, COUNT(r.id) AS rentals FROM member m
LEFT JOIN rental r ON r.member_id = m.id
GROUP BY m.id, m.name ORDER BY rentals DESC, m.id;
